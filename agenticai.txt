import os
import subprocess
import tempfile
from langchain_groq import ChatGroq
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

os.environ["GROQ_API_KEY"] = "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"

llm = ChatGroq(model="openai/gpt-oss-120b")

code_prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are an expert Python coding assistant. "
     "Generate only executable Python code. "
     "Do not use markdown. Do not explain."),
    ("user", "{task}")
])

fix_prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a self-correcting Python coding assistant. "
     "Fix the given code using the error message. "
     "Return only corrected executable Python code. "
     "Do not use markdown. Do not explain."),
    ("user",
     "Task:\n{task}\n\nCode:\n{code}\n\nError:\n{error}")
])

code_chain = code_prompt | llm | StrOutputParser()
fix_chain = fix_prompt | llm | StrOutputParser()


def clean_code(code):
    code = code.replace("```python", "").replace("```", "")
    return code.strip()


def run_python_code(code):
    with tempfile.NamedTemporaryFile(mode="w", suffix=".py", delete=False) as f:
        f.write(code)
        file_path = f.name

    result = subprocess.run(
        ["python", file_path],
        capture_output=True,
        text=True,
        timeout=10
    )

    if result.returncode == 0:
        return True, result.stdout
    else:
        return False, result.stderr


def self_correcting_coding_assistant(task, max_attempts=3):
    print("USER TASK:")
    print(task)
    print("\nGenerating code...\n")

    code = clean_code(code_chain.invoke({"task": task}))

    # Intentionally inject a bug for demonstration
    if "factorial" in task.lower():
      code = code.replace("factorial(5)", "fact(5)")
    for attempt in range(1, max_attempts + 1):
        print(f"\nAttempt {attempt}")
        print("-" * 40)
        print(code)

        success, output = run_python_code(code)

        if success:
            print("\nCode executed successfully!")
            print("\nOUTPUT:")
            print(output)
            return code

        print("\nError found:")
        print(output)

        print("\nCorrecting code...\n")
        code = clean_code(
            fix_chain.invoke({
                "task": task,
                "code": code,
                "error": output
            })
        )

    print("\nCould not fix the code within max attempts.")
    return code
final_code = self_correcting_coding_assistant(
    "Write Python code to calculate factorial of 5 using recursion."
)

from langgraph.graph import StateGraph,START,END
from langchain_groq import ChatGroq

llm=ChatGroq(model="openai/gpt-oss-120b")
class financestate(TypedDict):
    qsn:str
    stock:str
    price:float
    ans:str
def extractstock(state:financestate):
    qsn=state["qsn"].upper()
    if "APPLE" in qsn or "AAPL" in qsn:
        stock="AAPL"
    elif "TESLA" in qsn or "TSLA" in qsn:
        stock="TSLA"
    elif "MICROSOFT" in qsn or "MST" in qsn:
        stock="MST"
    else:
        stock="UNKOWN"
    return{"stock":stock}
def stockprice(state:financestate):
    fakeprice={
        "AAPL":80.0,
        "TSLA":120.3,
        "MST":23.4,
        "UNKOWN":0.0
    }
    price=fakeprice[state["stock"]]
    return {"price":price}
def answr(state:financestate):
    prompt=f"""
    you are finance assistant bot.
    userqs:{state['qsn']}
    ustock:{state['stock']}
    uprice:{state['price']}
    generate simple finacial response.
    Do not give investment advice.
    """
    response=llm.invoke(prompt)
    return{"ans":response.content}
graph=StateGraph(financestate)
graph.add_node("extractstock",extractstock)
graph.add_node("stockprice",stockprice)
graph.add_node("answr",answr)
graph.add_edge(START,"extractstock")
graph.add_edge("extractstock","stockprice")
graph.add_edge("stockprice","answr")
graph.add_edge("answr",END)

    Image(finance_bot.get_graph().draw_mermaid_png())
)

    "qsn":"what is current price of APPLE stock?"
})

import os
from typing import TypedDict
from langgraph.graph import StateGraph, END
from langchain_groq import ChatGroq
from IPython.display import Image, display

os.environ["GROQ_API_KEY"] ="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"

llm = ChatGroq(model="openai/gpt-oss-120b")

class State(TypedDict):
    query: str
    category: str
    response: str

def router(s):
    r = llm.invoke(
        f"Classify as faq, order or complaint. Return one word: {s['query']}"
    ).content.lower()
    return {
        "category": "order" if "order" in r
        else "complaint" if "complaint" in r else "faq"
    }

def faq(s):
    return {"response": llm.invoke(
        f"Answer politely: {s['query']}"
    ).content}

def order(s):
    return {"response": llm.invoke(
        f"Help with order. Ask for order ID if missing: {s['query']}"
    ).content}

def complaint(s):
    return {"response": llm.invoke(
        f"Apologize and suggest next step: {s['query']}"
    ).content}

def route(s):
    return {
        "faq": "FAQ",
        "order": "Order",
        "complaint": "Complaint"
    }[s["category"]]

g = StateGraph(State)

g.add_node("Router_Agent", router)
g.add_node("FAQ_Agent", faq)
g.add_node("Order_Agent", order)
g.add_node("Complaint_Agent", complaint)

g.set_entry_point("Router_Agent")

g.add_conditional_edges(
    "Router_Agent",
    route,
    {
        "FAQ": "FAQ_Agent",
        "Order": "Order_Agent",
        "Complaint": "Complaint_Agent"
    }
)

g.add_edge("FAQ_Agent", END)
g.add_edge("Order_Agent", END)
g.add_edge("Complaint_Agent", END)

app = g.compile()

# Generate graph
display(Image(app.get_graph().draw_mermaid_png()))

# Test chatbot
q = "I received a damaged product"

r = app.invoke({
    "query": q,
    "category": "",
    "response": ""
})

print("User Query:", q)
print("Category:", r["category"])
print("Bot Response:", r["response"])

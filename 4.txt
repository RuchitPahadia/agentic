import os
from langchain_google_genai import ChatGoogleGenerativeAI

#os.environ["GOOGLE_API_KEY"] = ""

llm = ChatGoogleGenerativeAI(
    model="gemini-3.5-flash-lite"
)

code = """
def add(a,b)
    reurn a+b
"""

check = str(llm.invoke(
    "Find errors in this Python code:\n" + code
).content)

response = llm.invoke(
    "Correct the code using these errors:\n"
    + check + "\nCode:\n" + code
).content

print("Corrected Code:")
print(response[0]["text"])
